% PPO算法框图
\begin{tikzpicture}[
    node distance=1.2cm,
    box/.style={rectangle, rounded corners, draw=blue!60, fill=blue!5,

    process/.style={rectangle, rounded corners, draw=orange!60, fill=orange!5,

    data/.style={rectangle, rounded corners, draw=green!60, fill=green!5,

% Input
\node[data] (env) at (0,0) {Environment};

\node[process] (update) at (0,-8.5) {PPO Update\\$L^{CLIP}(\theta)$};

% Arrows
\draw[arrow] (env) -- (actor);

\draw[arrow] (critic) -- (value);
